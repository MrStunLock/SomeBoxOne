cmake_minimum_required(VERSION 3.20)
project(SomeBoxOne VERSION 0.32.2 LANGUAGES CXX)

# An interpreter is several times slower without optimisation. Single-configuration generators (Ninja, Makefiles) used to default to an
# empty, unoptimised build type, so make Release the default there. Visual Studio picks the configuration at build time (--config Release).
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# Keep UTF-8 source literals consistent in MSVC builds (including Win32 GUI strings).
if(MSVC)
  add_compile_options(/utf-8 /validate-charset)
endif()

if(WIN32)
  link_libraries(winmm)   # waveOut audio output (src/audio_sink.hpp) and timeBeginPeriod
  enable_language(RC)
  configure_file(src/app_icon.rc.in ${CMAKE_CURRENT_BINARY_DIR}/SomeBoxOneAppIcon.rc @ONLY)
  set(SOMEBOXONE_ICON_RESOURCE ${CMAKE_CURRENT_BINARY_DIR}/SomeBoxOneAppIcon.rc)
endif()

add_executable(SomeBoxOne src/main.cpp ${SOMEBOXONE_ICON_RESOURCE})
target_include_directories(SomeBoxOne PRIVATE src)
if(MSVC)
  target_compile_options(SomeBoxOne PRIVATE /W4 /permissive-)
else()
  target_compile_options(SomeBoxOne PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
endif()

# A native Windows GUI: no Qt, .NET, or third-party runtime is required.
if(WIN32)
  # Standalone graphics window, separate from the main GUI process. It receives
  # captured title frames over a versioned shared-memory surface and presents them
  # with a host D3D11/DXGI swap chain (GDI fallback available).
  add_executable(SomeBoxOneDisplay WIN32 src/game_window_win32.cpp src/d3d11_presenter_win32.cpp ${SOMEBOXONE_ICON_RESOURCE})
  target_include_directories(SomeBoxOneDisplay PRIVATE src)
  target_link_libraries(SomeBoxOneDisplay PRIVATE user32 gdi32 ole32 windowscodecs d3d11 dxgi d3dcompiler)
  if(MSVC)
    target_compile_options(SomeBoxOneDisplay PRIVATE /W4 /permissive- /utf-8 /validate-charset)
  else()
    target_compile_options(SomeBoxOneDisplay PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
  endif()

  add_executable(SomeBoxOneGUI WIN32 src/gui_win32.cpp ${SOMEBOXONE_ICON_RESOURCE})
  target_include_directories(SomeBoxOneGUI PRIVATE src)
  target_link_libraries(SomeBoxOneGUI PRIVATE user32 comdlg32 shell32 ole32 comctl32)
  if(MSVC)
    target_compile_options(SomeBoxOneGUI PRIVATE /W4 /permissive-)
  else()
    target_compile_options(SomeBoxOneGUI PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
  endif()
endif()
